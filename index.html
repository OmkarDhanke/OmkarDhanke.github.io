revenue_analysis.sql

-- Identify revenue leakage by category
            WITH leakage_cte AS (
              SELECT
                customer_id,
                contract_type,
                billing_status,
                SUM(expected_revenue)
                  - SUM(actual_revenue)
                  AS leakage_amount
              FROM billing_data
              WHERE billing_status != 'PAID'
              GROUP BY 1, 2, 3
            )
            SELECT *,
              CASE
                WHEN leakage_amount > 1000
                THEN 'HIGH RISK'
                ELSE 'MONITOR'
              END AS risk_flag
            FROM leakage_cte
            ORDER BY leakage_amount DESC;
Query returned 247 rows

leakage_classification.sql 
-- Classify all leakage types
SELECT
  i.customer_id,
  i.invoice_amount,
  'Zombie Account'
    AS leakage_type,
  i.invoice_amount
    AS exposure
FROM invoices i
LEFT JOIN payments p
  ON i.invoice_id
    = p.invoice_id
WHERE p.payment_id
  IS NULL
  AND i.status
    = 'Open'

UNION ALL

SELECT
  customer_id,
  invoice_amount,
  'Partial Payment',
  invoice_amount
    - paid_amount
FROM payments
WHERE paid_amount
  < invoice_amount;
